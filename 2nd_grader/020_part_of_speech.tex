  \begin{frame}<1-5>[plain,label=basic]{品詞}
    \large
    \begin{description}
      \item<2->[名詞] 人、動物、植物やモノの名前を表す語\hfill\visible<3->{\small child, cat, flower,
        desk \ldots}

      \item<4->[代名詞] 名詞の代わりに使う語\hfill\visible<5->{\small I, we, you he, she, it, they,
        this, that \ldots}

      \item<6->[動詞] 動作や状態を表す語\hfill\visible<7->{\small run, see, speak, eat, study, have,
        be \ldots{}}

      \item<8->[形容詞] 名詞をくわしく説明する語\hfill\visible<9->{\small big, new, happy, good,
        beautiful \ldots}

      \item<10->[副詞] 動詞や形容詞などをくわしく説明する語\hfill\visible<11->{\small quickly, here,
        then, very \ldots}

      \item<12->[前置詞] 名詞や代名詞と組み合わせて$+\alpha$の意味をくわえる語\\ \hfill\visible<13->{\small
        in, on, under, with, before, after, from, to \ldots}

      \item<14->[冠詞] 名詞の前に置かれ、\kenten{特定}のものか\kenten{不特定}のものかを表す語\\
        \hfill\visible<15->{\small a, an, theだけ}

      \item<16->[接続詞] 複数の要素を接続する語\hfill\visible<17->{\small and, or, but}
    \end{description}

    \vfill

    \hfill\visible<18->{まずはこれをしっかり理解しましょう}

  \begin{frame}[plain]{冠詞}
    \large

    冠詞: a / an ,\hspace{10pt}the

    \begin{enumerate}
      \item $\text{a / an}+ \text{単数形の名詞}$\,(1つの～、ある～)\hfill{}$<$ one
        \begin{enumerate}
          \item a book\hfill\textipa{/\textschwa /} $+$ 子音

          \item an apple\hfill\textipa{/\textschwa n/} $+$ 母音
        \end{enumerate}

      \item $\text{the}+ \text{名詞}$\,(その～)\hspace{10pt}{\small (単数形・複数形どちらにもつく)}
        \begin{enumerate}
          \item the book / the books\hfill\textipa{/\dh\textschwa/} $+$ 子音

          \item the apple / the apples\hfill\textipa{/\dh i/} $+$ 母音
        \end{enumerate}
    \end{enumerate}

  \section{品詞と文の要素}
  %%%%%%%%%%%%%%%%%%%%%%
  \begin{frame}[plain]{品詞と文の要素}
    \begin{enumerate}
      %  \item<1-> birdの品詞は？\hfill\visible<2->{名詞}

      \item<1-> Birds sing.という英文のBirdsの品詞は？\hfill\visible<2->{Birdsは名詞}

      \item<3-> Birds sing. という英文の主語は？\hfill\visible<4->{Birdsは主語}
    \end{enumerate}

    \visible<5->{てことはBirds sing.のBirdsは名詞でもあるし主語でもあるってことなの？}

    \visible<6->{Birds sing.のBirdsは「名詞です」といったり「主語です」といったり、なんか混乱しちゃうんですけど}
  \end{frame}
  %%%%%%%%%%%%%%%%%%%%%%
  \begin{frame}[plain]{品詞と文の要素}
    説明の角度がちがいます\pause

    \bigskip

    「Birds sing.のBirdsは主語です」というときは、Birdsという単語が文の中でどういう役割をしているかについて言及しています。\pause

    おなじbirdsでもI love birds.という文のbirdsは目的語ですよね\pause

    \bigskip

  \begin{frame}[plain,label=basic]{Check Point}
    つぎの語の品詞を答えましょう

    \large
    \begin{description}
      \item[\visible<2->{名詞}] 人、動物、植物やモノの名前を表す語\hfill{\small child, cat, flower, desk \ldots}

      \item[\visible<3->{代名詞}] 名詞の代わりに使う語\hfill{\small I, we, you he, she, it, they, this, that \ldots}

      \item[\visible<4->{動詞}] 動作や状態を表す語\hfill{\small run, see, speak, eat, study, have, be \ldots{}}

      \item[\visible<5->{形容詞}] 名詞をくわしく説明する語\hfill{\small big, new, happy, good, beautiful \ldots}

      \item[\visible<6->{副詞}] 動詞や形容詞などをくわしく説明する語\hfill{\small quickly, here, then, very \ldots}

      \item[\visible<7->{前置詞}] 名詞や代名詞と組み合わせて$+\alpha$の意味をくわえる語\\ \hfill{\small in, on, under, with, before, after, from, to \ldots}

      \item[\visible<8->{冠詞}] 名詞の前に置かれ、\kenten{特定}のものか\kenten{不特定}のものかを表す語\\
        \hfill{\small a, an, theだけ}

      \item[\visible<9->{接続詞}] 複数の要素を接続する語\hfill{\small and, or, but}
    \end{description}
  \end{frame}
